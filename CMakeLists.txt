cmake_minimum_required(VERSION 3.16)
project(KotorSagaCP1C LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(kotor_graphics_settings src/GraphicsSettings.cpp src/GraphicsMenu.cpp src/FramePacer.cpp)
target_include_directories(kotor_graphics_settings PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
add_executable(kotor_settings_demo src/ConfigDemo.cpp)
target_link_libraries(kotor_settings_demo PRIVATE kotor_graphics_settings)
add_executable(kotor_settings_tests tests/SettingsTests.cpp)
target_link_libraries(kotor_settings_tests PRIVATE kotor_graphics_settings)
add_executable(kotor_host_logic_tests tests/HostLogicTests.cpp)
target_link_libraries(kotor_host_logic_tests PRIVATE kotor_graphics_settings)
enable_testing()
add_test(NAME GraphicsSettingsTests COMMAND kotor_settings_tests)
add_test(NAME HostLogicTests COMMAND kotor_host_logic_tests)
